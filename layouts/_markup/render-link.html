{{- /* 裸のリモートURLは data/linkcards.json のOGPキャッシュがあればリンクカードにする。 */ -}} {{- $link := .Destination -}} {{- $text := .Text -}} {{- $isRemote := findRE "^https?://" $link -}} {{- $isBare := eq $text $link -}} {{- $title := "" -}} {{- $desc := "" -}} {{- $image := "" -}} {{- if and $isRemote $isBare -}} {{- if isset hugo.Data "linkcards" -}} {{- with index hugo.Data.linkcards $link -}} {{- $title = .title -}} {{- $desc = .desc -}} {{- $image = .image -}} {{- end -}} {{- end -}} {{- end -}} {{- if $title -}} {{ $title }} {{ $desc | truncate 100 }} {{ $link | replaceRE "^https?://([^/]+).*" "$1" }} {{- if $image -}} {{- else -}} {{- end -}} {{- else if and $isRemote $isBare -}} {{ $text | safeHTML }} {{- else -}} {{ $text | safeHTML }} {{- end -}}
